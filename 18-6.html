② 智能客服系统 ER 实体关系图 erDiagram 6 实体 · 10 关系
用户 — 会话 — 消息 — 意图 — 知识库 — 工单 实体关系，核心数据模型
erDiagram
    %% ===== 用户 =====
    USER ||--o{ SESSION : "发起"
    USER {
        bigint user_id PK
        string nickname
        string avatar_url
        string phone
        string channel "来源渠道"
        datetime created_at
    }

    %% ===== 会话 =====
    SESSION ||--o{ MESSAGE : "包含"
    SESSION }o--|| AGENT : "可转人工"
    SESSION {
        bigint session_id PK
        bigint user_id FK
        string status "active|closed|transferred"
        int turn_count "轮次"
        string sentiment "情绪得分"
        datetime start_time
        datetime end_time
    }

    %% ===== 消息 =====
    MESSAGE }o--|| INTENT : "识别为"
    MESSAGE {
        bigint message_id PK
        bigint session_id FK
        int seq_num "顺序号"
        string role "user|assistant|system"
        text content
        float confidence "NLU 置信度"
        datetime created_at
    }

    %% ===== 意图 =====
    INTENT {
        bigint intent_id PK
        string intent_name "意图名：查订单/退换货..."
        string intent_type "类别：信息查询/业务办理"
        text description
        bigint faq_id FK "关联 FAQ"
    }

    %% ===== 知识库 =====
    FAQ ||--o{ INTENT : "对应"
    FAQ {
        bigint faq_id PK
        string question
        text answer
        string category "分类"
        int hit_count "命中次数"
        float quality_score "质量分"
        datetime updated_at
    }

    %% ===== 工单 =====
    SESSION ||--o{ TICKET : "生成"
    TICKET {
        bigint ticket_id PK
        bigint session_id FK
        bigint agent_id FK
        string status "pending|processing|resolved"
        string priority "priority"
        text issue_desc
        datetime created_at
    }

    %% ===== 人工客服 =====
    AGENT ||--o{ TICKET : "处理"
    AGENT {
        bigint agent_id PK
        string name
        string department
        string status "online|busy|offline"
        int current_tickets
    }
